find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui videoio)

if(NANOTRACKER_BUILD_ORT)
  set(ONNXRUNTIME_ROOT "" CACHE PATH "Extracted ONNX Runtime Linux GPU package")
  find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include")
  find_library(ONNXRUNTIME_LIBRARY onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib")
  if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
    message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to an extracted ONNX Runtime 1.30 GPU package.")
  endif()

  add_executable(nanotracker_cpp src/main.cpp)
  target_compile_features(nanotracker_cpp PRIVATE cxx_std_20)
  set_target_properties(nanotracker_cpp PROPERTIES CXX_EXTENSIONS OFF BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
  target_compile_options(nanotracker_cpp PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(nanotracker_cpp PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
  target_link_libraries(nanotracker_cpp PRIVATE "${ONNXRUNTIME_LIBRARY}" ${OpenCV_LIBS})
  target_compile_definitions(nanotracker_cpp PRIVATE NANOTRACKER_MODELS_DIR="${PROJECT_SOURCE_DIR}/models/nanotrackv3")
endif()

if(NANOTRACKER_BUILD_RKNN)
  find_path(RKNN_INCLUDE_DIR rknn_api.h HINTS "${RKNN_ROOT}/include")
  find_library(RKNN_LIBRARY rknnrt HINTS "${RKNN_ROOT}/aarch64")
  if(NOT RKNN_INCLUDE_DIR OR NOT RKNN_LIBRARY)
    message(FATAL_ERROR "Set RKNN_ROOT to rknpu2/runtime/Linux/librknn_api")
  endif()
  add_executable(nanotracker_rknn src/rknn_main.cpp)
  target_compile_features(nanotracker_rknn PRIVATE cxx_std_20)
  set_target_properties(nanotracker_rknn PROPERTIES CXX_EXTENSIONS OFF BUILD_RPATH "${RKNN_ROOT}/aarch64")
  target_compile_options(nanotracker_rknn PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(nanotracker_rknn PRIVATE "${RKNN_INCLUDE_DIR}")
  target_link_libraries(nanotracker_rknn PRIVATE "${RKNN_LIBRARY}" ${OpenCV_LIBS})
  target_compile_definitions(nanotracker_rknn PRIVATE NANOTRACKER_RKNN_MODELS_DIR="${PROJECT_SOURCE_DIR}/models/nanotrackv3_rknn_rk3566")
endif()

if(NANOTRACKER_BUILD_NCNN)
  set(NCNN_ROOT "" CACHE PATH "Extracted NCNN Linux shared package")
  find_package(ncnn CONFIG REQUIRED PATHS "${NCNN_ROOT}/lib/cmake/ncnn" NO_DEFAULT_PATH)

  add_executable(nanotracker_ncnn src/ncnn_main.cpp)
  target_compile_features(nanotracker_ncnn PRIVATE cxx_std_20)
  set_target_properties(nanotracker_ncnn PROPERTIES CXX_EXTENSIONS OFF BUILD_RPATH "${NCNN_ROOT}/lib")
  target_compile_options(nanotracker_ncnn PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(nanotracker_ncnn PRIVATE ncnn ${OpenCV_LIBS})
  target_compile_definitions(nanotracker_ncnn PRIVATE NANOTRACKER_MODELS_DIR="${PROJECT_SOURCE_DIR}/models/nanotrackv3_ncnn")
endif()
